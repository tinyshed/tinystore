// explain prints which b-tree operations SQLite compiles for the metrics
// ingest UPDATE shapes, against the series_state schema of migrations 0001-0003
package main

import (
	"database/sql"
	"fmt"
	"os"
	"path/filepath"
	"runtime"
	"sort"
	"strings"

	_ "modernc.org/sqlite"
)

func main() {
	dir, err := os.MkdirTemp("", "explain")
	if err != nil {
		panic(err)
	}
	defer os.RemoveAll(dir)
	db, err := sql.Open("sqlite", filepath.Join(dir, "probe.db"))
	if err != nil {
		panic(err)
	}
	defer db.Close()
	for _, statement := range []string{
		`create table series_state (series_id integer primary key, max_seen_ts integer not null, sealed_before integer, version integer not null default 0, head_count integer not null default 0, ready integer not null default 0, next_gc_ts integer, model_scale integer not null default -2, tail blob, head_start integer, head_end integer) strict`,
		`create index series_ready on series_state(series_id) where ready = 1`,
		`create index series_due on series_state(next_gc_ts) where next_gc_ts is not null`,
	} {
		if _, err = db.Exec(statement); err != nil {
			panic(err)
		}
	}
	var version string
	if err = db.QueryRow(`select sqlite_version()`).Scan(&version); err != nil {
		panic(err)
	}
	fmt.Printf("%s %s/%s, SQLite %s\n", runtime.Version(), runtime.GOOS, runtime.GOARCH, version)
	shapes := map[string]string{
		"1 ingest state (ingest.go:145)": `update series_state set max_seen_ts=max(max_seen_ts,?),version=version+1,ready=?,next_gc_ts=case when next_gc_ts is null then ? else min(next_gc_ts,?) end where series_id=?`,
		"2 head tail (head.go:222)":      `update series_state set tail=?,head_count=?,head_start=?,head_end=? where series_id=?`,
		"3 merged, index columns unset":  `update series_state set tail=?,head_count=?,head_start=?,head_end=?,max_seen_ts=max(max_seen_ts,?),version=version+1 where series_id=?`,
		"4 merged, index columns set":    `update series_state set tail=?,head_count=?,head_start=?,head_end=?,max_seen_ts=max(max_seen_ts,?),version=version+1,ready=?,next_gc_ts=? where series_id=?`,
	}
	names := make([]string, 0, len(shapes))
	for name := range shapes {
		names = append(names, name)
	}
	sort.Strings(names)
	for _, name := range names {
		query := shapes[name]
		rows, err := db.Query(`explain `+query, make([]any, strings.Count(query, "?"))...)
		if err != nil {
			panic(err)
		}
		var ops []string
		for rows.Next() {
			var address, p1, p2, p3, p5 int
			var opcode, p4, comment sql.NullString
			if err = rows.Scan(&address, &opcode, &p1, &p2, &p3, &p4, &p5, &comment); err != nil {
				panic(err)
			}
			if strings.HasPrefix(opcode.String, "Idx") || opcode.String == "Insert" || opcode.String == "Delete" {
				ops = append(ops, opcode.String)
			}
		}
		if err = rows.Err(); err != nil {
			panic(err)
		}
		rows.Close()
		fmt.Printf("%-32s %s\n", name, strings.Join(ops, " "))
	}
}
