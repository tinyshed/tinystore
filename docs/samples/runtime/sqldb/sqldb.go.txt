// Package sqldb gives an application its own SQLite databases inside the store.
// You write the schema and the SQL; TinyStore owns the file, the connections,
// the migrations and the transaction lifecycle.
package sqldb

import (
	"context"
	"crypto/sha256"
	"database/sql"
	"errors"
	"fmt"
	"io/fs"
	"log/slog"
	"net/url"
	"path/filepath"
	"regexp"
	"slices"

	_ "modernc.org/sqlite"

	"github.com/tinyshed/tinystore"
)

var (
	ErrManyRows = errors.New("sqldb: more than one row")
	ErrShape    = errors.New("sqldb: columns do not fit the destination")
	ErrSchema   = errors.New("sqldb: schema does not match the migrations")
)

// a name becomes a file name, so it stays short and plain
var validName = regexp.MustCompile(`^[a-z0-9][a-z0-9_-]{0,63}$`)

type DB struct {
	name string
	db   *sql.DB
	log  *slog.Logger
}

// Open opens or creates sql/<name>.db and applies, in one transaction, every
// migration it has not applied before:
//
//	001_users.sql   applied on the first run
//	002_posts.sql   applied the first time a binary carrying it opens the file
//
// A migration changed after it was applied, or a database newer than the
// binary's migrations, refuses to open.
func Open(ctx context.Context, store *tinystore.Store, name string, migrations fs.FS) (*DB, error) {
	if !validName.MatchString(name) {
		return nil, fmt.Errorf("%w: database name %q", tinystore.ErrInvalid, name)
	}
	path, err := store.Claim("sql/" + name + ".db")
	if err != nil {
		return nil, err
	}

	pool, err := sql.Open("sqlite", fileURL(path))
	if err != nil {
		return nil, fmt.Errorf("sql %q: open: %w", name, err)
	}
	db := &DB{name: name, db: pool, log: store.Logger("sql").With("database", name)}

	if err := db.migrate(ctx, migrations); err != nil {
		return nil, errors.Join(err, pool.Close())
	}
	if err := store.Attach(db); err != nil {
		return nil, errors.Join(err, pool.Close())
	}
	return db, nil
}

func (d *DB) Exec(ctx context.Context, query string, args ...any) (sql.Result, error) {
	return d.db.ExecContext(ctx, query, args...)
}

func (d *DB) Query(ctx context.Context, query string, args ...any) (*sql.Rows, error) {
	return d.db.QueryContext(ctx, query, args...)
}

func (d *DB) QueryRow(ctx context.Context, query string, args ...any) *sql.Row {
	return d.db.QueryRowContext(ctx, query, args...)
}

// One reads exactly one row: none is sql.ErrNoRows, two are ErrManyRows.
func (d *DB) One[T any](ctx context.Context, query string, args ...any) (T, error) {
	return one[T](ctx, d.db, query, args...)
}

func (d *DB) All[T any](ctx context.Context, query string, args ...any) ([]T, error) {
	return all[T](ctx, d.db, query, args...)
}

// Scalar reads one row of one column, such as a count or a returning id.
func (d *DB) Scalar[T any](ctx context.Context, query string, args ...any) (T, error) {
	return scalar[T](ctx, d.db, query, args...)
}

// Transaction runs work in one transaction: nil commits, an error or a panic
// rolls back.
func (d *DB) Transaction(ctx context.Context, work func(*Tx) error) (err error) {
	tx, err := d.db.BeginTx(ctx, nil)
	if err != nil {
		return fmt.Errorf("sql %q: begin: %w", d.name, err)
	}
	defer func() {
		// after a commit this is a no-op; after an error or a panic it undoes the work
		if rollbackErr := tx.Rollback(); rollbackErr != nil && !errors.Is(rollbackErr, sql.ErrTxDone) {
			err = errors.Join(err, fmt.Errorf("sql %q: rollback: %w", d.name, rollbackErr))
		}
	}()

	if err = work(&Tx{tx: tx}); err != nil {
		return err
	}
	if err = tx.Commit(); err != nil {
		return fmt.Errorf("sql %q: commit: %w", d.name, err)
	}
	return nil
}

func (d *DB) Close(context.Context) error {
	d.log.Info("database closed")
	return d.db.Close()
}

// Tx has the same reads and writes as DB, inside one transaction.
type Tx struct{ tx *sql.Tx }

func (t *Tx) Exec(ctx context.Context, query string, args ...any) (sql.Result, error) {
	return t.tx.ExecContext(ctx, query, args...)
}

func (t *Tx) Query(ctx context.Context, query string, args ...any) (*sql.Rows, error) {
	return t.tx.QueryContext(ctx, query, args...)
}

func (t *Tx) One[T any](ctx context.Context, query string, args ...any) (T, error) {
	return one[T](ctx, t.tx, query, args...)
}

func (t *Tx) All[T any](ctx context.Context, query string, args ...any) ([]T, error) {
	return all[T](ctx, t.tx, query, args...)
}

func (t *Tx) Scalar[T any](ctx context.Context, query string, args ...any) (T, error) {
	return scalar[T](ctx, t.tx, query, args...)
}

type appliedMigration struct {
	Version  int
	Name     string
	Checksum []byte
}

func (d *DB) migrate(ctx context.Context, migrations fs.FS) error {
	names, err := fs.Glob(migrations, "*.sql")
	if err != nil {
		return fmt.Errorf("sql %q: list migrations: %w", d.name, err)
	}
	slices.Sort(names)

	return d.Transaction(ctx, func(tx *Tx) error {
		_, err := tx.Exec(ctx, `create table if not exists _tinystore_migrations (
			version  integer primary key,
			name     text not null,
			checksum blob not null
		) strict`)
		if err != nil {
			return fmt.Errorf("sql %q: migration history: %w", d.name, err)
		}

		applied, err := tx.All[appliedMigration](ctx, `select version, name, checksum from _tinystore_migrations order by version`)
		if err != nil {
			return fmt.Errorf("sql %q: migration history: %w", d.name, err)
		}
		if len(applied) > len(names) {
			return fmt.Errorf("%w: sql %q is at %d, this binary knows %d", ErrSchema, d.name, len(applied), len(names))
		}

		for i, name := range names {
			body, err := fs.ReadFile(migrations, name)
			if err != nil {
				return fmt.Errorf("sql %q: read %s: %w", d.name, name, err)
			}
			sum := sha256.Sum256(body)

			if i < len(applied) {
				if applied[i].Name != name || string(applied[i].Checksum) != string(sum[:]) {
					return fmt.Errorf("%w: sql %q: %s changed after it was applied", ErrSchema, d.name, name)
				}
				continue
			}

			if _, err = tx.Exec(ctx, string(body)); err != nil {
				return fmt.Errorf("sql %q: apply %s: %w", d.name, name, err)
			}
			_, err = tx.Exec(ctx, `insert into _tinystore_migrations values (?, ?, ?)`, i+1, name, sum[:])
			if err != nil {
				return fmt.Errorf("sql %q: record %s: %w", d.name, name, err)
			}
			d.log.Info("migration applied", "migration", name)
		}
		return nil
	})
}

func fileURL(path string) string {
	absolute, err := filepath.Abs(path)
	if err != nil {
		absolute = path
	}
	arguments := url.Values{"_txlock": {"immediate"}}
	for _, pragma := range []string{"journal_mode(WAL)", "busy_timeout(5000)", "foreign_keys(1)"} {
		arguments.Add("_pragma", pragma)
	}
	return (&url.URL{Scheme: "file", Path: filepath.ToSlash(absolute), RawQuery: arguments.Encode()}).String()
}
