// Package tinystore keeps an application's data in one directory: a file per
// engine, and one lifecycle for all of them.
//
//	data/
//	├── metrics.db     metrics.Open(ctx, store, …)
//	├── records.db     records.Open(ctx, store, …)
//	├── jobs.db        jobs.Open(ctx, store, …)
//	├── blobs/         blobs.Open(ctx, store, …)
//	└── sql/
//	    └── app.db     sqldb.Open(ctx, store, "app", migrations)
//
// Open the store, then each engine the application uses against it. Close
// stops background work and closes every engine, the last opened first. The
// store imports no engine, so a program links only the engines it opens.
package tinystore

import (
	"context"
	"errors"
	"fmt"
	"log/slog"
	"os"
	"path/filepath"
	"slices"
	"sync"
	"time"
)

var (
	ErrInvalid = errors.New("tinystore: invalid request")
	ErrClosed  = errors.New("tinystore: store is closed")
	ErrInUse   = errors.New("tinystore: name already in use")
)

type Options struct {
	// Logger receives TinyStore's own logs; nil discards them.
	Logger *slog.Logger

	// Manual stops all background work: the application calls each engine's
	// Maintain itself. Tests usually want this.
	Manual bool

	// SelfMetrics, when set, writes the store's own numbers into an opened
	// metrics engine at this interval.
	SelfMetrics time.Duration

	// Clock replaces time.Now.
	Clock func() time.Time
}

type Store struct {
	dir    string
	logger *slog.Logger
	clock  func() time.Time
	manual bool

	background context.Context
	stop       context.CancelFunc
	running    sync.WaitGroup

	mu        sync.Mutex
	closed    bool
	claimed   map[string]bool
	engines   []Engine
	reporters []Reporter
	sink      SelfSink
}

// Engine is what the store needs from an opened engine.
type Engine interface {
	Close(ctx context.Context) error
}

// Reporter is an engine that can describe its own work as numbers.
type Reporter interface {
	Report() []Measure
}

// SelfSink keeps the store's own measurements; the metrics engine is one.
type SelfSink interface {
	WriteSelf(ctx context.Context, at time.Time, measures []Measure) error
}

// Measure is one number an engine reports about itself, like
// {Engine: "metrics", Name: "ingested_samples", Value: 1200}.
type Measure struct {
	Engine, Name string
	Value        float64
}

func Open(ctx context.Context, dir string, options Options) (*Store, error) {
	if dir == "" {
		return nil, fmt.Errorf("%w: empty directory", ErrInvalid)
	}
	if err := os.MkdirAll(dir, 0o750); err != nil {
		return nil, fmt.Errorf("tinystore: create %s: %w", dir, err)
	}

	store := &Store{
		dir:     dir,
		logger:  options.Logger,
		clock:   options.Clock,
		manual:  options.Manual,
		claimed: map[string]bool{},
	}
	if store.logger == nil {
		store.logger = slog.New(slog.DiscardHandler)
	}
	if store.clock == nil {
		store.clock = time.Now
	}
	store.background, store.stop = context.WithCancel(context.WithoutCancel(ctx))

	if options.SelfMetrics > 0 {
		store.Every("self metrics", options.SelfMetrics, store.writeSelfMetrics)
	}
	store.logger.Info("store opened", "dir", dir)
	return store, nil
}

// Close stops background work, then closes every engine, the last opened first.
func (s *Store) Close(ctx context.Context) error {
	s.mu.Lock()
	if s.closed {
		s.mu.Unlock()
		return nil
	}
	s.closed = true
	engines := s.engines
	s.mu.Unlock()

	s.stop()
	s.running.Wait()

	var err error
	for _, engine := range slices.Backward(engines) {
		err = errors.Join(err, engine.Close(ctx))
	}
	s.logger.Info("store closed", "dir", s.dir)
	return err
}

// The methods below are how an engine joins the store.

// Claim reserves a file or directory for one engine and returns its path.
// Engine names and application databases cannot collide:
//
//	metrics.db  records.db  jobs.db  blobs/     engines
//	sql/<name>.db                              databases the application names
func (s *Store) Claim(name string) (string, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	if s.closed {
		return "", ErrClosed
	}
	if s.claimed[name] {
		return "", fmt.Errorf("%w: %s", ErrInUse, name)
	}
	s.claimed[name] = true

	path := filepath.Join(s.dir, filepath.FromSlash(name))
	if err := os.MkdirAll(filepath.Dir(path), 0o750); err != nil {
		return "", fmt.Errorf("tinystore: create %s: %w", filepath.Dir(path), err)
	}
	return path, nil
}

// Attach hands an opened engine to the store: Close will close it, and the
// store reads its numbers or writes its own through it when it can.
func (s *Store) Attach(engine Engine) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	if s.closed {
		return ErrClosed
	}
	s.engines = append(s.engines, engine)
	if reporter, ok := engine.(Reporter); ok {
		s.reporters = append(s.reporters, reporter)
	}
	if sink, ok := engine.(SelfSink); ok && s.sink == nil {
		s.sink = sink
	}
	return nil
}

// Logger is the store's logger, labelled with the engine that logs.
func (s *Store) Logger(engine string) *slog.Logger {
	return s.logger.With("engine", engine)
}

func (s *Store) Now() time.Time {
	return s.clock()
}

// Every runs work at each interval until the store closes. A Manual store runs
// nothing, and a failure is logged rather than stopping the next run.
func (s *Store) Every(name string, interval time.Duration, work func(context.Context) error) {
	if s.manual {
		return
	}
	s.running.Go(func() {
		ticker := time.NewTicker(interval)
		defer ticker.Stop()
		for {
			select {
			case <-s.background.Done():
				return
			case <-ticker.C:
			}
			if err := work(s.background); err != nil && !errors.Is(err, context.Canceled) {
				s.logger.Warn("background work failed", "work", name, "error", err)
			}
		}
	})
}

func (s *Store) writeSelfMetrics(ctx context.Context) error {
	s.mu.Lock()
	reporters, sink := slices.Clone(s.reporters), s.sink
	s.mu.Unlock()
	if sink == nil {
		return nil
	}

	var measures []Measure
	for _, reporter := range reporters {
		measures = append(measures, reporter.Report()...)
	}
	return sink.WriteSelf(ctx, s.clock(), measures)
}
