// Package records keeps structured log records in records.db. Its Handler
// plugs into log/slog, so an application's logs land here as well as wherever
// else the application already sends them.
package records

import (
	"context"
	"database/sql"
	"encoding/json"
	"errors"
	"fmt"
	"log/slog"
	"net/url"
	"path/filepath"
	"slices"
	"sync"
	"sync/atomic"
	"time"

	_ "modernc.org/sqlite"

	"github.com/tinyshed/tinystore"
)

type Options struct {
	// Buffer is how many records may wait in memory; beyond it a record is
	// dropped and counted, never waited for. 1024 when zero.
	Buffer int

	// Flush is how often waiting records are written. One second when zero.
	Flush time.Duration
}

type Record struct {
	At      time.Time
	Level   slog.Level
	Message string
	Attrs   map[string]any
}

type Store struct {
	db      *sql.DB
	log     *slog.Logger
	queue   chan Record
	flushMu sync.Mutex
	written atomic.Int64
	dropped atomic.Int64
}

func Open(ctx context.Context, store *tinystore.Store, options Options) (*Store, error) {
	if options.Buffer == 0 {
		options.Buffer = 1024
	}
	if options.Flush == 0 {
		options.Flush = time.Second
	}

	path, err := store.Claim("records.db")
	if err != nil {
		return nil, err
	}
	db, err := sql.Open("sqlite", fileURL(path))
	if err != nil {
		return nil, fmt.Errorf("records: open: %w", err)
	}
	_, err = db.ExecContext(ctx, `create table if not exists records (
		at      integer not null,
		level   integer not null,
		message text    not null,
		attrs   text    not null
	) strict`)
	if err != nil {
		return nil, errors.Join(fmt.Errorf("records: schema: %w", err), db.Close())
	}

	engine := &Store{db: db, log: store.Logger("records"), queue: make(chan Record, options.Buffer)}
	if err := store.Attach(engine); err != nil {
		return nil, errors.Join(err, db.Close())
	}
	store.Every("records flush", options.Flush, engine.Flush)
	return engine, nil
}

// Handler queues records for records.db without ever blocking the caller.
func (s *Store) Handler() slog.Handler {
	return &handler{store: s}
}

// Flush writes every waiting record in one transaction.
func (s *Store) Flush(ctx context.Context) (err error) {
	s.flushMu.Lock()
	defer s.flushMu.Unlock()

	batch := s.drain()
	if len(batch) == 0 {
		return nil
	}

	tx, err := s.db.BeginTx(ctx, nil)
	if err != nil {
		return fmt.Errorf("records: begin: %w", err)
	}
	defer func() {
		if rollbackErr := tx.Rollback(); rollbackErr != nil && !errors.Is(rollbackErr, sql.ErrTxDone) {
			err = errors.Join(err, fmt.Errorf("records: rollback: %w", rollbackErr))
		}
	}()
	for _, record := range batch {
		attrs, err := json.Marshal(record.Attrs)
		if err != nil {
			return fmt.Errorf("records: encode attributes: %w", err)
		}
		_, err = tx.ExecContext(ctx, `insert into records values (?, ?, ?, ?)`,
			record.At.UnixMilli(), int(record.Level), record.Message, string(attrs))
		if err != nil {
			return fmt.Errorf("records: write: %w", err)
		}
	}
	if err := tx.Commit(); err != nil {
		return fmt.Errorf("records: commit: %w", err)
	}

	s.written.Add(int64(len(batch)))
	s.log.Debug("records written", "count", len(batch))
	return nil
}

// Since returns the records at or after a moment, oldest first.
func (s *Store) Since(ctx context.Context, since time.Time) ([]Record, error) {
	rows, err := s.db.QueryContext(ctx, `select at, level, message, attrs from records where at >= ? order by at`, since.UnixMilli())
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var out []Record
	for rows.Next() {
		var at int64
		var level int
		var record Record
		var attrs string
		if err := rows.Scan(&at, &level, &record.Message, &attrs); err != nil {
			return nil, err
		}
		record.At, record.Level = time.UnixMilli(at), slog.Level(level)
		if err := json.Unmarshal([]byte(attrs), &record.Attrs); err != nil {
			return nil, err
		}
		out = append(out, record)
	}
	return out, rows.Err()
}

func (s *Store) Report() []tinystore.Measure {
	return []tinystore.Measure{
		{Engine: "records", Name: "written_records", Value: float64(s.written.Load())},
		{Engine: "records", Name: "dropped_records", Value: float64(s.dropped.Load())},
	}
}

// Close writes what is still waiting, then closes records.db.
func (s *Store) Close(ctx context.Context) error {
	return errors.Join(s.Flush(ctx), s.db.Close())
}

func (s *Store) drain() []Record {
	var batch []Record
	for {
		select {
		case record := <-s.queue:
			batch = append(batch, record)
		default:
			return batch
		}
	}
}

// handler turns slog records into Records. The records engine's own log lines
// are refused, or writing them would log again, forever.
type handler struct {
	store  *Store
	attrs  []slog.Attr
	prefix string
	own    bool
}

func (h *handler) Enabled(context.Context, slog.Level) bool {
	return !h.own
}

func (h *handler) Handle(_ context.Context, r slog.Record) error {
	if h.own {
		return nil
	}
	attrs := make(map[string]any, len(h.attrs)+r.NumAttrs())
	for _, attr := range h.attrs {
		attrs[attr.Key] = attr.Value.Resolve().Any()
	}
	r.Attrs(func(attr slog.Attr) bool {
		attrs[h.prefix+attr.Key] = attr.Value.Resolve().Any()
		return true
	})

	select {
	case h.store.queue <- Record{At: r.Time, Level: r.Level, Message: r.Message, Attrs: attrs}:
	default:
		h.store.dropped.Add(1)
	}
	return nil
}

func (h *handler) WithAttrs(attrs []slog.Attr) slog.Handler {
	next := *h
	next.attrs = slices.Clone(h.attrs)
	for _, attr := range attrs {
		next.attrs = append(next.attrs, slog.Attr{Key: h.prefix + attr.Key, Value: attr.Value})
		if attr.Key == "engine" && attr.Value.String() == "records" {
			next.own = true
		}
	}
	return &next
}

func (h *handler) WithGroup(name string) slog.Handler {
	next := *h
	next.prefix = h.prefix + name + "."
	return &next
}

func fileURL(path string) string {
	absolute, err := filepath.Abs(path)
	if err != nil {
		absolute = path
	}
	arguments := url.Values{"_txlock": {"immediate"}}
	for _, pragma := range []string{"journal_mode(WAL)", "busy_timeout(5000)"} {
		arguments.Add("_pragma", pragma)
	}
	return (&url.URL{Scheme: "file", Path: filepath.ToSlash(absolute), RawQuery: arguments.Encode()}).String()
}
